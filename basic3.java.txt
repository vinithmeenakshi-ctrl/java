import java.util.Scanner;

public class GroceryStoreSales {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

       
        double sales = sc.nextDouble();
        double discountPercent = sc.nextDouble();
        double gstPercent = sc.nextDouble();


        double discountAmount = sales * discountPercent / 100;
        double amountAfterDiscount = sales - discountAmount;

       
        double gstAmount = amountAfterDiscount * gstPercent / 100;
        double finalRevenue = amountAfterDiscount + gstAmount;

        System.out.println("Final Revenue = " + finalRevenue);

        sc.close();
    }
}
