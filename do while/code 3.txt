import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int num = sc.nextInt();

        int reverse = 0;

        do {
            int digit = num % 10;
            reverse = reverse * 10 + digit;
            num /= 10;
        } while(num != 0);

        System.out.println("Reversed = " + reverse);
    }
}
